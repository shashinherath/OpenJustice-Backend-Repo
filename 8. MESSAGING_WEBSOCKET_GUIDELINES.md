## 🔐 WebSocket Authentication Rules

Authenticate WebSocket connections before allowing any message exchange. Use JWT tokens passed during connection handshake or in the first message. Never allow unauthenticated WebSocket connections. Validate tokens on connection and periodically refresh to prevent token expiration issues.

### JWT-Based WebSocket Auth

```python
# app/api/websockets/auth.py
from fastapi import WebSocket, WebSocketDisconnect, status
from jose import jwt, JWTError
from app.core.security import SECRET_KEY, ALGORITHM

async def authenticate_websocket(websocket: WebSocket) -> int:
    """
    Authenticate WebSocket connection using JWT.

    Expects token in query parameter: ws://host/chat?token=<jwt>

    Returns:
        int: Authenticated user_id

    Raises:
        WebSocketDisconnect: If authentication fails
    """

    # Get token from query parameters
    token = websocket.query_params.get('token')

    if not token:
        logger.warning(f"WebSocket connection attempt without token from {websocket.client}")
        await websocket.close(code=status.WS_1008_POLICY_VIOLATION)
        raise WebSocketDisconnect(code=status.WS_1008_POLICY_VIOLATION)

    try:
        # Decode JWT
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        user_id = payload.get('sub')

        if user_id is None:
            raise JWTError("Invalid token payload")

        logger.info(f"WebSocket authenticated for user {user_id}")
        return int(user_id)

    except JWTError as e:
        logger.warning(f"WebSocket auth failed: {e}")
        await websocket.close(code=status.WS_1008_POLICY_VIOLATION)
        raise WebSocketDisconnect(code=status.WS_1008_POLICY_VIOLATION)
```

## 🔄 Connection Lifecycle Management

Manage WebSocket connections centrally to track active connections, handle disconnections gracefully, and broadcast messages efficiently. Implement connection cleanup on disconnection or timeout. Maintain per-user connection lists to enable targeted messaging and prevent memory leaks.

### Connection Manager

```python
# app/core/websocket_manager.py
from typing import Dict, List
from fastapi import WebSocket
import asyncio

class ConnectionManager:
    """Manage WebSocket connections"""

    def __init__(self):
        # user_id -> list of WebSocket connections
        self.active_connections: Dict[int, List[WebSocket]] = {}
        self._lock = asyncio.Lock()

    async def connect(self, user_id: int, websocket: WebSocket):
        """Register new connection"""
        async with self._lock:
            if user_id not in self.active_connections:
                self.active_connections[user_id] = []

            self.active_connections[user_id].append(websocket)

            logger.info(f"User {user_id} connected (total connections: {len(self.active_connections[user_id])})")

    async def disconnect(self, user_id: int, websocket: WebSocket = None):
        """Remove connection"""
        async with self._lock:
            if user_id in self.active_connections:
                if websocket:
                    # Remove specific connection
                    self.active_connections[user_id].remove(websocket)
                else:
                    # Remove all connections for user
                    self.active_connections[user_id] = []

                # Clean up empty lists
                if not self.active_connections[user_id]:
                    del self.active_connections[user_id]

                logger.info(f"User {user_id} disconnected")

    async def send_personal_message(self, user_id: int, message: dict):
        """Send message to specific user (all their connections)"""
        if user_id in self.active_connections:
            # Send to all user's connections (multiple devices)
            for connection in self.active_connections[user_id]:
                try:
                    await connection.send_json(message)
                except Exception as e:
                    logger.error(f"Failed to send message to user {user_id}: {e}")
                    await self.disconnect(user_id, connection)

    async def broadcast(self, message: dict, exclude_user: int = None):
        """Broadcast message to all connected users"""
        for user_id, connections in self.active_connections.items():
            if user_id != exclude_user:
                for connection in connections:
                    try:
                        await connection.send_json(message)
                    except Exception as e:
                        logger.error(f"Failed to broadcast to user {user_id}: {e}")

    def get_connection_count(self) -> int:
        """Get total number of active connections"""
        return sum(len(conns) for conns in self.active_connections.values())

    def is_user_connected(self, user_id: int) -> bool:
        """Check if user is connected"""
        return user_id in self.active_connections and len(self.active_connections[user_id]) > 0

# Global instance
connection_manager = ConnectionManager()
```

## 🚦 Rate Limiting

Apply rate limiting to WebSocket messages to prevent abuse and DoS attacks. Track messages per user per time window. Disconnect or throttle users exceeding rate limits. Use sliding window algorithms for fair enforcement across burst and sustained traffic.

### WebSocket Rate Limiter

```python
# app/core/rate_limiter.py
from collections import defaultdict
from datetime import datetime, timedelta
import asyncio

class WebSocketRateLimiter:
    """Rate limit WebSocket messages"""

    def __init__(
        self,
        max_messages: int = 10,
        window_seconds: int = 60
    ):
        self.max_messages = max_messages
        self.window_seconds = window_seconds

        # user_id -> list of message timestamps
        self.message_history: Dict[int, List[datetime]] = defaultdict(list)
        self._lock = asyncio.Lock()

    async def is_allowed(self, user_id: int) -> bool:
        """Check if user is allowed to send message"""

        async with self._lock:
            now = datetime.utcnow()
            cutoff = now - timedelta(seconds=self.window_seconds)

            # Remove old timestamps
            self.message_history[user_id] = [
                ts for ts in self.message_history[user_id]
                if ts > cutoff
            ]

            # Check limit
            if len(self.message_history[user_id]) >= self.max_messages:
                logger.warning(f"Rate limit exceeded for user {user_id}")
                return False

            # Add current timestamp
            self.message_history[user_id].append(now)
            return True

    async def get_remaining(self, user_id: int) -> int:
        """Get remaining messages in current window"""
        async with self._lock:
            now = datetime.utcnow()
            cutoff = now - timedelta(seconds=self.window_seconds)

            recent = [
                ts for ts in self.message_history[user_id]
                if ts > cutoff
            ]

            return max(0, self.max_messages - len(recent))

# Global instance
rate_limiter = WebSocketRateLimiter(max_messages=10, window_seconds=60)
```

Integrate rate limiting into message handlers before processing any user input. Check limits synchronously and reject messages immediately if limits are exceeded. Provide clear error messages to users about rate limits.

            "type": "error",

            "message": f"Unknown message type: {message_type}"
        })

## 📬 Message Ordering Guarantees

Maintain message ordering for chat conversations to ensure coherent user experience. Assign sequence numbers to messages and process them in order. Handle out-of-order messages by queueing or rejecting them. This is critical for multi-turn conversations where context depends on message order.

### Message Queue with Sequencing

```python
# app/core/message_queue.py
from collections import deque
from dataclasses import dataclass
from typing import Deque

@dataclass
class QueuedMessage:
    """Message with sequence number"""
    sequence: int
    user_id: int
    content: dict
    timestamp: datetime

class MessageQueue:
    """Ensure message ordering"""

    def __init__(self):
        # user_id -> deque of messages
        self.queues: Dict[int, Deque[QueuedMessage]] = defaultdict(deque)
        self.sequence_counters: Dict[int, int] = defaultdict(int)
        self._lock = asyncio.Lock()

    async def enqueue(self, user_id: int, content: dict) -> int:
        """Add message to queue with sequence number"""

        async with self._lock:
            # Get next sequence number
            sequence = self.sequence_counters[user_id]
            self.sequence_counters[user_id] += 1

            # Create queued message
            message = QueuedMessage(
                sequence=sequence,
                user_id=user_id,
                content=content,
                timestamp=datetime.utcnow()
            )

            # Add to queue
            self.queues[user_id].append(message)

            logger.debug(f"Message queued for user {user_id}: seq={sequence}")
            return sequence

    async def dequeue(self, user_id: int) -> QueuedMessage:
        """Get next message from queue"""

        async with self._lock:
            if user_id in self.queues and self.queues[user_id]:
                message = self.queues[user_id].popleft()
                return message
            return None

    async def peek(self, user_id: int) -> QueuedMessage:
        """Peek at next message without removing"""

        async with self._lock:
            if user_id in self.queues and self.queues[user_id]:
                return self.queues[user_id][0]
            return None
```

### Message Processing with Ordering

Process messages sequentially per connection or conversation. Queue messages and process them in sequence number order. Detect and handle gaps in sequence numbers to prevent data loss or corruption.

```python
async def process_chat_message(user_id: int, data: dict, websocket: WebSocket):
    """Process chat message with ordering guarantee"""

    # Enqueue message
    sequence = await message_queue.enqueue(user_id, data)

    # Process messages in order
    while True:
        next_message = await message_queue.peek(user_id)

        if not next_message:
            break

        try:
            # Process message
            response = await chat_service.process_message(
                user_id=next_message.user_id,
                message=next_message.content['message']
            )

            # Send response with sequence number
            await websocket.send_json({
                "type": "chat_response",
                "sequence": next_message.sequence,
                "response": response,
                "timestamp": datetime.utcnow().isoformat()
            })

            # Remove from queue
            await message_queue.dequeue(user_id)

        except Exception as e:
            logger.error(f"Message processing error: {e}")

            # Send error response
            await websocket.send_json({
                "type": "error",
                "sequence": next_message.sequence,
                "error": str(e)
            })

            # Remove failed message
            await message_queue.dequeue(user_id)
            break
```

## 📱 WhatsApp Webhook Idempotency

WhatsApp and other webhook providers may send duplicate events due to retries or network issues. Implement idempotency to handle duplicate webhooks safely. Use message IDs or webhook IDs to detect and skip already-processed events. Store processed IDs with TTL to save memory.

### Webhook Deduplication

```python
# app/services/whatsapp_service.py
from typing import Set
import hashlib

class WhatsAppWebhookHandler:
    """Handle WhatsApp webhooks with idempotency"""

    def __init__(self):
        # Store processed webhook IDs (consider using Redis for production)
        self.processed_webhooks: Set[str] = set()
        self._lock = asyncio.Lock()

        # Time-to-live for processed IDs (cleanup old ones)
        self.ttl_seconds = 3600  # 1 hour

    def generate_webhook_id(self, webhook_data: dict) -> str:
        """Generate unique ID for webhook"""

        # Use message ID if available
        if 'entry' in webhook_data:
            for entry in webhook_data['entry']:
                for change in entry.get('changes', []):
                    messages = change.get('value', {}).get('messages', [])
                    if messages:
                        return messages[0]['id']

        # Fallback: Hash entire payload
        payload_str = json.dumps(webhook_data, sort_keys=True)
        return hashlib.sha256(payload_str.encode()).hexdigest()

    async def is_duplicate(self, webhook_id: str) -> bool:
        """Check if webhook already processed"""

        async with self._lock:
            if webhook_id in self.processed_webhooks:
                logger.warning(f"Duplicate webhook detected: {webhook_id}")
                return True

            # Mark as processed
            self.processed_webhooks.add(webhook_id)

            # Schedule removal after TTL
            asyncio.create_task(self._remove_after_ttl(webhook_id))

            return False

    async def _remove_after_ttl(self, webhook_id: str):
        """Remove webhook ID after TTL"""
        await asyncio.sleep(self.ttl_seconds)

        async with self._lock:
            self.processed_webhooks.discard(webhook_id)
            logger.debug(f"Webhook ID expired: {webhook_id}")

# Global instance
webhook_handler = WhatsAppWebhookHandler()
```

## 🔄 Retry & Failure Handling

Implement robust retry logic for failed message deliveries with exponential backoff. Distinguish between transient failures (network errors, timeouts) and permanent failures (invalid recipients, authorization errors). Use dead letter queues for messages that fail after max retries.

### Exponential Backoff for Failed Messages

```python
async def send_with_retry(
    websocket: WebSocket,
    message: dict,
    max_retries: int = 3
):
    """Send message with retry logic"""

    for attempt in range(max_retries):
        try:
            await websocket.send_json(message)
            return True

        except WebSocketDisconnect:
            logger.warning("WebSocket disconnected during send")
            return False

        except Exception as e:
            logger.warning(f"Send attempt {attempt + 1} failed: {e}")

            if attempt < max_retries - 1:
                # Exponential backoff
                wait_time = 2 ** attempt
                await asyncio.sleep(wait_time)

    logger.error("Failed to send message after all retries")
    return False
```

### Dead Letter Queue

Move messages to a dead letter queue after exceeding retry limits. Monitor DLQ for patterns indicating systemic issues. Implement manual review and reprocessing workflows for DLQ messages.

```python
# app/core/dead_letter_queue.py

class DeadLetterQueue:
    """Store failed messages for later retry"""

    def __init__(self):
        self.failed_messages: List[dict] = []
        self._lock = asyncio.Lock()

    async def add(self, user_id: int, message: dict, error: str):
        """Add failed message to DLQ"""

        async with self._lock:
            self.failed_messages.append({
                "user_id": user_id,
                "message": message,
                "error": error,
                "timestamp": datetime.utcnow(),
                "retry_count": 0
            })

            logger.info(f"Message added to DLQ for user {user_id}")

    async def retry_failed_messages(self):
        """Retry messages in DLQ"""

        async with self._lock:
            to_retry = self.failed_messages.copy()
            self.failed_messages.clear()

        for item in to_retry:
            try:
                # Check if user is connected
                if connection_manager.is_user_connected(item['user_id']):
                    await connection_manager.send_personal_message(
                        item['user_id'],
                        item['message']
                    )
                    logger.info(f"DLQ message retried for user {item['user_id']}")
                else:
                    # User still not connected, add back to DLQ
                    if item['retry_count'] < 5:
                        item['retry_count'] += 1
                        async with self._lock:
                            self.failed_messages.append(item)

            except Exception as e:
                logger.error(f"DLQ retry failed: {e}")
```

## 🚨 Common Mistakes to Avoid

❌ **No authentication**

```python
# BAD
@app.websocket("/ws")
async def websocket(ws: WebSocket):
    await ws.accept()  # Anyone can connect!
```

❌ **No rate limiting**

```python
# BAD
while True:
    data = await ws.receive_json()
    await process(data)  # No limit!
```

❌ **Ignoring duplicates**

```python
# BAD
@app.post("/webhook")
async def webhook(data: dict):
    process(data)  # Processes duplicates!
```

❌ **No error handling**

```python
# BAD
await websocket.send_json(msg)  # No try-except!
```
